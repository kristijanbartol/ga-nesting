\title{Pattern Layout Optimization for \\ Seam-Aware Texture Alignment}

\section{Formal Problem Specification: Seam Topology and Texture-Aware Nesting}

\subsection{Problem Domain and Inputs}

We define the problem as
\[
\Psi = \langle M, \mathcal{T}, \mathcal{L}, \Omega \rangle,
\]
where:

\begin{itemize}
    \item $M = (V,E,F)$ is the 3D garment mesh.
    \item $\mathcal{T}$ is a periodic texture defined over the fabric plane.
    \item The periodicity is characterized by a lattice basis $B = \{\mathbf{u}, \mathbf{v}\}$.
    \item $\mathcal{L} = \{L_1, \dots, L_m\}$ defines admissible regions for seam keypoints.
    \item $\Omega = [0,W] \times [0,\infty)$ is the fabric domain.
\end{itemize}

\subsection{Decision Variables (Genotype)}

A candidate solution is

\[
\mathbf{x} =
\langle
\boldsymbol{\delta},
\boldsymbol{\rho},
\boldsymbol{\kappa},
\mathbf{w},
\pi,
h
\rangle.
\]

\subsubsection{Seam Topology ($\boldsymbol{\delta}$)}

Each seam keypoint $k_j$ is defined as
\[
k_j = \Gamma_j(\delta_j), \quad \delta_j \in [0,1],
\]
where $\Gamma_j$ maps into region $L_j \subset M$.

Seams $s \in \mathcal{S}$ are geodesics between corresponding keypoints.

\subsubsection{Fabric Orientation ($\boldsymbol{\rho}$)}

After cutting along seams, we obtain charts
\[
P(\boldsymbol{\delta}) = \{p_1, \dots, p_n\}.
\]

Each chart has discrete grain orientation
\[
\rho_i \in \{0,1,2,3\}, \quad \theta_i = \rho_i \frac{\pi}{2}.
\]

\subsubsection{Discrete Texture Phase Offsets ($\boldsymbol{\kappa}$)}

To model periodic texture freedom, each chart is assigned a discrete phase bin.

Let $K$ be the number of phase bins per period.

\[
\kappa_i \in \{0,1,\dots,K-1\}.
\]

The effective phase shift for chart $p_i$ is
\[
\frac{\kappa_i}{K}.
\]

This represents the discrete repeat offset within the fabric's fundamental period.

\subsubsection{Seam Weights ($\mathbf{w}$)}

For each internal seam $s \in \mathcal{S}_{adj}$,
\[
w_s \in [0,1]
\]
defines its relative importance.

\subsubsection{Nesting Variables ($\pi, h$)}

\begin{itemize}
    \item $\pi \in S_n$ defines placement order.
    \item $h \in \mathcal{H}$ selects the nesting heuristic.
\end{itemize}

\subsection{Phenotype: Layout Construction}

For each chart $p_i$:

\begin{itemize}
    \item Independently flatten $p_i$ to $\phi_i: p_i \to \mathbf{R}^2$.
    \item Apply rotation $R(\theta_i)$.
    \item Determine translation $\mathbf{t}_i \in \Omega$ via nesting heuristic $h$ in order $\pi$.
\end{itemize}

Charts must satisfy

\begin{align}
(\phi_i(p_i) + \mathbf{t}_i) \cap (\phi_j(p_j) + \mathbf{t}_j) &= \emptyset, \\
(\phi_i(p_i) + \mathbf{t}_i) &\subset \Omega.
\end{align}

\subsection{Texture Phase Evaluation}

Let $\mathrm{Phase}(\mathbf{x})$ denote the lattice phase of fabric position $\mathbf{x}$,
computed using basis $B$ and reduced modulo one period.

For chart $p_i$, the effective phase at fabric point $\mathbf{x}$ is

\[
\mathrm{Phase}_i(\mathbf{x}) =
\mathrm{Phase}(\mathbf{x}) +
\frac{\kappa_i}{K}.
\]

Phase is evaluated modulo one.

\paragraph{Wrapped Phase Difference}

Given two phases $\phi_a$ and $\phi_b$ in $[0,1)$,
their wrapped difference is defined as

\[
\Delta(\phi_a, \phi_b)
=
\min\left(
|\phi_a - \phi_b|,
1 - |\phi_a - \phi_b|
\right).
\]

\subsection{Seam Phase Mismatch}

For seam $s$ connecting charts $p_i$ and $p_j$,
let $l$ parameterize corresponding seam points.

The seam phase mismatch is defined as

\[
\mathrm{Mismatch}(s)
=
\frac{1}{L_s}
\int_0^{L_s}
\Delta(
\mathrm{Phase}_i(\mathbf{x}_i(l)),
\mathrm{Phase}_j(\mathbf{x}_j(l))
)
\, dl.
\]

\subsection{Objective Functions}

The fitness vector is

\[
\mathbf{F}(\mathbf{x}) = [f_1, f_2, f_3]^T.
\]

\paragraph{Fabric Efficiency}

\[
f_1(\mathbf{x}) =
\max_{p \in \text{Layout}} y_{max}(p).
\]

\paragraph{Texture Phase Discontinuity}

\[
f_2(\mathbf{x}) =
\sum_{s \in \mathcal{S}_{adj}}
w_s \cdot \mathrm{Mismatch}(s).
\]

\paragraph{Flattening Distortion}

\[
f_3(\mathbf{x}) =
\sum_{i=1}^n E_{iso}(\phi_i).
\]
